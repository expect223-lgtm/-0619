        .active-scale:active { transform: scale(0.96); }
    </style>

<body class="bg-gray-100 min-h-screen font-sans pb-36">

    <div class="bg-indigo-600 text-white p-5 shadow-md sticky top-0 z-50 text-center">
        <h1 class="text-sm font-semibold tracking-wider text-indigo-200 uppercase mb-1">📅 本次會議/活動點名日期</h1>
        <div class="max-w-xs mx-auto">
            <input id="rollCallDate" type="date" class="w-full text-center bg-indigo-700 text-white border-none rounded-lg px-4 py-2.5 text-xl font-bold focus:outline-none focus:ring-2 focus:ring-indigo-300 shadow-inner">
        </div>
    </div>

    <div class="max-w-md mx-auto p-4 space-y-4">
        
        <div class="bg-white rounded-xl shadow-sm p-4 border border-gray-200">
            <h2 class="text-base font-bold text-gray-700 mb-3 flex items-center gap-1">
                <span>👤 人員管理 (新增/修改)</span>
            </h2>
            <div class="grid grid-cols-2 gap-2 mb-2">
                <input id="inputUnit" type="text" placeholder="單位 (如: 業務部)" class="border border-gray-300 rounded-lg px-3 py-3 text-base focus:outline-none focus:ring-2 focus:ring-indigo-500 bg-gray-50">
                <input id="inputName" type="text" placeholder="姓名 (如: 王小明)" class="border border-gray-300 rounded-lg px-3 py-3 text-base focus:outline-none focus:ring-2 focus:ring-indigo-500 bg-gray-50">
            </div>
            <button onclick="addOrUpdatePerson()" class="w-full bg-indigo-500 text-white py-3 rounded-lg font-bold text-base active-scale shadow-sm">
                ➕ 儲存/更新至本地名冊
            </button>
        </div>

        <div class="bg-white rounded-xl shadow-sm p-4 border border-gray-200">
            <div class="flex justify-between items-center mb-4 border-b pb-2">
                <h2 class="text-base font-bold text-gray-800">📋 本次會議出席名冊</h2>
                <button onclick="clearAll()" class="text-xs text-red-500 border border-red-200 px-2 py-1 rounded-md bg-red-50">重設名冊</button>
            </div>
            
            <div id="customStatusBlock" class="hidden mb-4 p-3 bg-yellow-50 rounded-lg border border-yellow-200 text-sm">
                <p class="font-semibold text-yellow-700 mb-1.5">💡 請輸入新增的出席狀態名稱：</p>
                <div class="flex gap-2">
                    <input id="customStatusInput" type="text" placeholder="例如: 公假、請假、遲到" class="flex-1 border rounded-lg px-3 py-2">
                    <button onclick="saveCustomStatus()" class="bg-yellow-600 text-white px-4 py-2 rounded-lg font-medium">確定</button>
                </div>
            </div>

            <div id="personList" class="space-y-4">
                <p class="text-gray-400 text-center py-8 text-sm">目前名冊無人，請在上方新增人員。</p>
            </div>
        </div>

        <div class="pt-2">
            <button onclick="submitToCloud()" id="submitBtn" class="w-full bg-green-600 text-white py-4 rounded-xl font-black text-lg active-scale shadow-lg tracking-wide">
                📤 上傳此場會議紀錄至 Google 試算表
            </button>
        </div>

    </div>

    <div class="fixed bottom-0 left-0 right-0 bg-white border-t border-gray-200 shadow-2xl p-4 max-w-md mx-auto z-40 rounded-t-xl">
        <div class="flex justify-around items-center text-center">
            <div>
                <div id="statRate" class="text-3xl font-black text-indigo-600">0%</div>
                <div class="text-xs text-gray-400 font-bold mt-0.5">本次出席率</div>
            </div>
            <div class="border-l h-10 border-gray-200"></div>
            <div>
                <div id="statPresent" class="text-xl font-bold text-green-600">0</div>
                <div class="text-xs text-gray-500">出席</div>
            </div>
            <div>
                <div id="statAbsent" class="text-xl font-bold text-red-500">0</div>
                <div class="text-xs text-gray-500">缺席</div>
            </div>
            <div>
                <div id="statOther" class="text-xl font-bold text-yellow-600">0</div>
                <div class="text-xs text-gray-500">其他原因</div>
            </div>
        </div>
    </div>

    <script>
        // ⚠️請在此處貼上你從 Google Apps Script 複製的網頁應用程式網址
        const GAS_WEB_APP_URL = "https://script.google.com/macros/s/AKfycbz4k0ki0hPT2aHkj7owbMH3Tk353k2UJQjHU-RamONYgISn8DH7kfXor_H1OyERgnJoUg/exec";

        // 初始化資料庫
        let people = JSON.parse(localStorage.getItem('conf_people')) || [];
        let currentCustomStatus = localStorage.getItem('conf_custom_status') || "其他";
        
        // 預設當天日期 (最上方獨立欄位)
        const dateInput = document.getElementById('rollCallDate');
        const today = new Date().toISOString().split('T')[0];
        dateInput.value = today;

        // 儲存至手機本地快取
        function saveToLocal() {
            localStorage.setItem('conf_people', JSON.stringify(people));
            localStorage.setItem('conf_custom_status', currentCustomStatus);
            calculateStats();
        }

        // 新增或修改人員
        function addOrUpdatePerson() {
            const unit = document.getElementById('inputUnit').value.trim();
            const name = document.getElementById('inputName').value.trim();
            
            if (!unit || !name) return alert('請填寫完整單位與姓名！');
            
            const existingIndex = people.findIndex(p => p.name === name);
            if (existingIndex > -1) {
                people[existingIndex].unit = unit; // 名字相同則更新單位
            } else {
                people.push({ unit, name, status: '出席', note: '' });
            }
            
            document.getElementById('inputUnit').value = '';
            document.getElementById('inputName').value = '';
            saveToLocal();
            renderList();
        }

        // 修改人員填回上方輸入框
        function editPerson(index) {
            document.getElementById('inputUnit').value = people[index].unit;
            document.getElementById('inputName').value = people[index].name;
            window.scrollTo({ top: 0, behavior: 'smooth' }); // 捲動到頂部方便修改
        }

        // 刪除人員
        function deletePerson(index) {
            if(confirm(`確定要將 ${people[index].name} 從名冊移除嗎？`)) {
                people.splice(index, 1);
                saveToLocal();
                renderList();
            }
        }

        // 變更出席狀態
        function setStatus(index, status) {
            if (status === 'other_setup') {
                document.getElementById('customStatusBlock').classList.remove('hidden');
                window.currentSettingIndex = index;
                return;
            }
            people[index].status = status;
            saveToLocal();
            renderList();
        }

        // 儲存自訂其他原因
        function saveCustomStatus() {
            const val = document.getElementById('customStatusInput').value.trim();
            if(val) {
                currentCustomStatus = val;
                if(window.currentSettingIndex !== undefined) {
                    people[window.currentSettingIndex].status = val;
                }
            }
            document.getElementById('customStatusBlock').classList.add('hidden');
            document.getElementById('customStatusInput').value = '';
            saveToLocal();
            renderList();
        }

        // 變更備註內容
        function updateNote(index, val) {
            people[index].note = val;
            localStorage.setItem('conf_people', JSON.stringify(people));
        }

        // 計算出席率
        function calculateStats() {
            if (people.length === 0) {
                document.getElementById('statRate').innerText = "0%";
                document.getElementById('statPresent').innerText = 0;
                document.getElementById('statAbsent').innerText = 0;
                document.getElementById('statOther').innerText = 0;
                return;
            }
            let present = 0, absent = 0, other = 0;
            people.forEach(p => {
                if (p.status === '出席') present++;
                else if (p.status === '缺席') absent++;
                else other++;
            });
            
            const rate = Math.round((present / people.length) * 100);
            document.getElementById('statRate').innerText = `${rate}%`;
            document.getElementById('statPresent').innerText = present;
            document.getElementById('statAbsent').innerText = absent;
            document.getElementById('statOther').innerText = other;
        }

        // 清除名冊
        function clearAll() {
            if(confirm('確定要清空所有人員名冊嗎？')) {
                people = [];
                saveToLocal();
                renderList();
            }
        }

        // 渲染畫面列表
        function renderList() {
            const container = document.getElementById('personList');
            if(people.length === 0) {
                container.innerHTML = '<p class="text-gray-400 text-center py-8 text-sm">目前名冊無人，請在上方新增人員。</p>';
                return;
            }

            container.innerHTML = people.map((p, index) => {
                const isOther = p.status !== '出席' && p.status !== '缺席';
                return `
                    <div class="bg-gray-50 p-3.5 rounded-xl border border-gray-200 space-y-2.5">
                        <div class="flex justify-between items-center">
                            <div>
                                <span class="text-xs bg-indigo-100 text-indigo-700 px-2 py-0.5 rounded font-bold">${p.unit}</span>
                                <span class="text-lg font-bold text-gray-800 ml-2">${p.name}</span>
                            </div>
                            <div class="flex gap-3 text-sm text-gray-400">
                                <button onclick="editPerson(${index})" class="hover:text-blue-600 font-medium">✏️修改</button>
                                <button onclick="deletePerson(${index})" class="hover:text-red-600 font-medium">❌刪除</button>
                            </div>
                        </div>
                        
                        <div class="grid grid-cols-3 gap-2">
                            <button onclick="setStatus(${index}, '出席')" class="py-3.5 rounded-xl font-bold text-base border active-scale ${p.status === '出席' ? 'bg-green-500 text-white border-green-500 shadow' : 'bg-white text-gray-600 border-gray-300'}">出席</button>
                            <button onclick="setStatus(${index}, '缺席')" class="py-3.5 rounded-xl font-bold text-base border active-scale ${p.status === '缺席' ? 'bg-red-500 text-white border-red-500 shadow' : 'bg-white text-gray-600 border-gray-300'}">缺席</button>
                            <button onclick="setStatus(${index}, 'other_setup')" class="py-3.5 rounded-xl font-bold text-sm border active-scale ${isOther ? 'bg-yellow-500 text-white border-yellow-500 shadow' : 'bg-white text-gray-600 border-gray-300'}">
                                ${isOther ? p.status : '+' + currentCustomStatus}
                            </button>
                        </div>

                        <input type="text" placeholder="💬 輸入特殊狀況備註 (如: 晚到10分鐘)..." value="${p.note}" onchange="updateNote(${index}, this.value)" class="w-full text-sm bg-white border border-gray-300 rounded-lg px-3 py-2.5 focus:outline-none focus:ring-2 focus:ring-indigo-500 text-gray-700">
                    </div>
                `;
            }).join('');
        }

        // 上傳資料到雲端 Google Sheets
        function submitToCloud() {
            if (people.length === 0) return alert('名冊內無人可點名！');
            if (GAS_WEB_APP_URL.includes("請在此處貼上")) return alert('尚未設定 GAS 雲端網址，無法上傳！');
            
            const submitBtn = document.getElementById('submitBtn');
            submitBtn.innerText = "⏳ 正在傳輸新會議紀錄...";
            submitBtn.disabled = true;

            // 抓取最上方選定的會議日期
            const selectedDate = dateInput.value;
            
            const payload = {
                action: "saveRollCall",
                data: people.map(p => ({
                    date: selectedDate,
                    unit: p.unit,
                    name: p.name,
                    status: p.status,
                    note: p.note
                }))
            };

            fetch(GAS_WEB_APP_URL, {
                method: "POST",
                mode: "no-cors", 
                cache: "no-cache",
                headers: { "Content-Type": "application/json" },
                body: JSON.stringify(payload)
            })
            .then(() => {
                alert(`🎉 ${selectedDate} 會議點名成功！資料已追加至雲端試算表末端。`);
                submitBtn.innerText = "📤 上傳此場會議紀錄至 Google 試算表";
                submitBtn.disabled = false;
            })
            .catch(err => {
                alert('❌ 上傳失敗：' + err);
                submitBtn.innerText = "📤 上傳此場會議紀錄至 Google 試算表";
                submitBtn.disabled = false;
            });
        }

        // 初始載入
        renderList();
        calculateStats();
    </script>
